<script>
    const DB_KEY = 'MuscleTrackerDB_v2';

    const MUSCLE_GROUPS = [
        { id: 'chest', name: 'Гърди' },
        { id: 'back', name: 'Гръб' },
        { id: 'legs', name: 'Крака' },
        { id: 'shoulders', name: 'Рамене' },
        { id: 'arms', name: 'Ръце' },
        { id: 'core', name: 'Корем' }
    ];

    let currentChartPoints = [];
    let selectedPastGroups = new Set();

    function getTodayString() {
        return formatDate(new Date());
    }

    function formatDate(dateObj) {
        return dateObj.getFullYear() + '-' +
               String(dateObj.getMonth() + 1).padStart(2, '0') + '-' +
               String(dateObj.getDate()).padStart(2, '0');
    }

    function getDefaultDatabase() {
        const today = getTodayString();
        return {
            dailyGainRateGrams: 15.5,
            lastAutoCheckDate: today,
            entries: [
                {
                    date: today,
                    trained: false,
                    gainedGrams: 15.5,
                    groups: []
                }
            ]
        };
    }

    function loadDB() {
        const raw = localStorage.getItem(DB_KEY);
        if (!raw) {
            const initial = getDefaultDatabase();
            saveDB(initial);
            return initial;
        }
        const db = JSON.parse(raw);
        // Миграция на стари записи, ако нямат масив groups
        db.entries.forEach(e => {
            if (!Array.isArray(e.groups)) {
                e.groups = [];
            }
        });
        return db;
    }

    function saveDB(db) {
        db.entries.sort((a, b) => a.date.localeCompare(b.date));
        localStorage.setItem(DB_KEY, JSON.stringify(db));
        renderUI(db);
    }

    function checkAndApplyDailyUpdate() {
        const db = loadDB();
        const today = getTodayString();

        if (db.entries.length === 0) {
            db.entries.push({
                date: today,
                trained: false,
                gainedGrams: parseFloat(db.dailyGainRateGrams),
                groups: []
            });
            saveDB(db);
            return;
        }

        db.entries.sort((a, b) => a.date.localeCompare(b.date));
        const latestEntryDateStr = db.entries[db.entries.length - 1].date;

        if (latestEntryDateStr < today) {
            let curr = new Date(latestEntryDateStr + 'T00:00:00');
            const end = new Date(today + 'T00:00:00');

            while (curr < end) {
                curr.setDate(curr.getDate() + 1);
                const dateStr = formatDate(curr);
                if (!db.entries.some(e => e.date === dateStr)) {
                    db.entries.push({
                        date: dateStr,
                        trained: Boolean(false),
                        gainedGrams: parseFloat(db.dailyGainRateGrams),
                        groups: []
                    });
                }
            }
            db.lastAutoCheckDate = today;
            saveDB(db);
        } else {
            renderUI(db);
        }
    }

    // Връща текущия (днешния или последния симулиран) запис
    function getCurrentEntry(db) {
        const today = getTodayString();
        let entry = db.entries.find(e => e.date === today);
        if (!entry && db.entries.length > 0) {
            entry = db.entries[db.entries.length - 1];
        }
        return entry;
    }

    // Превключване на конкретна мускулна група за днес
    function toggleTodayMuscleGroup(groupId) {
        const db = loadDB();
        const entry = getCurrentEntry(db);
        if (!entry) return;

        if (!Array.isArray(entry.groups)) entry.groups = [];
        const idx = entry.groups.indexOf(groupId);

        if (idx === -1) {
            entry.groups.push(groupId);
            entry.trained = true; // Автоматично правим bool = true
        } else {
            entry.groups.splice(idx, 1);
            if (entry.groups.length === 0) {
                entry.trained = false;
            }
        }
        saveDB(db);
    }

    // Общ бутон за тренировка днес
    function toggleTodayWorkout() {
        const db = loadDB();
        const entry = getCurrentEntry(db);
        if (!entry) return;

        entry.trained = !Boolean(entry.trained);
        if (!entry.trained) {
            entry.groups = [];
        }
        saveDB(db);
    }

    // Инициализиране на чиповете за избор на мускулни групи при минали дни
    function renderPastGroupChips() {
        const container = document.getElementById('pastGroupChips');
        container.innerHTML = '';
        MUSCLE_GROUPS.forEach(g => {
            const span = document.createElement('span');
            span.className = 'chip' + (selectedPastGroups.has(g.id) ? ' selected' : '');
            span.innerText = g.name;
            span.onclick = () => {
                if (selectedPastGroups.has(g.id)) {
                    selectedPastGroups.delete(g.id);
                } else {
                    selectedPastGroups.add(g.id);
                    document.getElementById('pastBoolInput').value = 'true';
                }
                renderPastGroupChips();
            };
            container.appendChild(span);
        });
    }

    // Когато потребителят избере дата от календара, зареждаме нейните данни ако вече съществуват
    function onPastDateChange() {
        const dateVal = document.getElementById('pastDateInput').value;
        const db = loadDB();
        const existing = db.entries.find(e => e.date === dateVal);
        if (existing) {
            document.getElementById('pastGramsInput').value = existing.gainedGrams;
            document.getElementById('pastBoolInput').value = existing.trained ? 'true' : 'false';
            selectedPastGroups = new Set(existing.groups || []);
            renderPastGroupChips();
        }
    }

    function addOrUpdatePastDay() {
        const dateVal = document.getElementById('pastDateInput').value;
        const gramsVal = parseFloat(document.getElementById('pastGramsInput').value);
        let boolVal = document.getElementById('pastBoolInput').value === 'true';
        const fillRange = document.getElementById('fillRangeCheckbox').checked;

        if (!dateVal) {
            alert('Моля, избери дата от календара!');
            return;
        }
        if (isNaN(gramsVal) || gramsVal < 0) {
            alert('Моля, въведи валиден грамаж (double)!');
            return;
        }

        const chosenGroups = Array.from(selectedPastGroups);
        if (chosenGroups.length > 0) {
            boolVal = true;
        }

        const db = loadDB();
        const roundedGrams = parseFloat(gramsVal.toFixed(2));

        const existingIndex = db.entries.findIndex(e => e.date === dateVal);
        if (existingIndex !== -1) {
            db.entries[existingIndex].gainedGrams = roundedGrams;
            db.entries[existingIndex].trained = Boolean(boolVal);
            db.entries[existingIndex].groups = chosenGroups;
        } else {
            db.entries.push({
                date: dateVal,
                trained: Boolean(boolVal),
                gainedGrams: roundedGrams,
                groups: chosenGroups
            });
        }

        if (fillRange) {
            let curr = new Date(dateVal + 'T00:00:00');
            const today = new Date(getTodayString() + 'T00:00:00');

            while (curr < today) {
                curr.setDate(curr.getDate() + 1);
                const dStr = formatDate(curr);
                if (!db.entries.some(e => e.date === dStr)) {
                    db.entries.push({
                        date: dStr,
                        trained: Boolean(boolVal),
                        gainedGrams: roundedGrams,
                        groups: [...chosenGroups]
                    });
                }
            }
        }

        saveDB(db);
    }

    function toggleWorkoutByDate(dateStr) {
        const db = loadDB();
        const entry = db.entries.find(e => e.date === dateStr);
        if (entry) {
            entry.trained = !Boolean(entry.trained);
            if (!entry.trained) entry.groups = [];
            saveDB(db);
        }
    }

    function loadDateIntoEditor(dateStr) {
        document.getElementById('pastDateInput').value = dateStr;
        onPastDateChange();
        document.getElementById('pastDateInput').scrollIntoView({ behavior: 'smooth', block: 'center' });
    }

    function deleteEntryByDate(dateStr) {
        const db = loadDB();
        if (db.entries.length <= 1) {
            alert('Базата трябва да съдържа поне 1 ден!');
            return;
        }
        db.entries = db.entries.filter(e => e.date !== dateStr);
        saveDB(db);
    }

    function updateDailyRate() {
        const inputVal = parseFloat(document.getElementById('inputDailyRate').value);
        if (isNaN(inputVal) || inputVal < 0) {
            alert('Моля, въведи валидно число (double)!');
            return;
        }
        const db = loadDB();
        db.dailyGainRateGrams = parseFloat(inputVal.toFixed(2));

        const latest = db.entries[db.entries.length - 1];
        if (latest) {
            latest.gainedGrams = db.dailyGainRateGrams;
        }
        saveDB(db);
    }

    function simulateNextDay() {
        const db = loadDB();
        db.entries.sort((a, b) => a.date.localeCompare(b.date));
        const lastDateStr = db.entries[db.entries.length - 1].date;
        const nextDate = new Date(lastDateStr + 'T00:00:00');
        nextDate.setDate(nextDate.getDate() + 1);

        db.entries.push({
            date: formatDate(nextDate),
            trained: Boolean(false),
            gainedGrams: parseFloat(db.dailyGainRateGrams),
            groups: []
        });

        saveDB(db);
    }

    function exportDatabase() {
        const db = loadDB();
        const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(db, null, 2));
        const downloadAnchor = document.createElement('a');
        downloadAnchor.setAttribute("href", dataStr);
        downloadAnchor.setAttribute("download", "muscle_database.json");
        document.body.appendChild(downloadAnchor);
        downloadAnchor.click();
        downloadAnchor.remove();
    }

    // --- РИСУВАНЕ НА CANVAS ГРАФИКАТА ---
    function drawChart(rows, filterKey) {
        const canvas = document.getElementById('progressCanvas');
        if (!canvas) return;
        const ctx = canvas.getContext('2d');

        const dpr = window.devicePixelRatio || 1;
        const rect = canvas.getBoundingClientRect();
        canvas.width = rect.width * dpr;
        canvas.height = rect.height * dpr;
        ctx.scale(dpr, dpr);

        const width = rect.width;
        const height = rect.height;
        const padLeft = 58;
        const padRight = 24;
        const padTop = 20;
        const padBottom = 38;

        const plotWidth = width - padLeft - padRight;
        const plotHeight = height - padTop - padBottom;

        ctx.clearRect(0, 0, width, height);
        if (rows.length === 0) return;

        const getValue = (r) => filterKey === 'all' ? r.cumulativeGrams : (r.groupCumulative[filterKey] || 0);

        const maxVal = Math.max(...rows.map(getValue), 5);
        const minVal = 0;

        const gridSteps = 4;
        ctx.font = '12px Segoe UI, sans-serif';
        ctx.textAlign = 'right';
        ctx.textBaseline = 'middle';

        for (let i = 0; i <= gridSteps; i++) {
            const ratio = i / gridSteps;
            const y = padTop + plotHeight - (ratio * plotHeight);
            const val = (minVal + ratio * (maxVal - minVal)).toFixed(1);

            ctx.strokeStyle = '#334155';
            ctx.lineWidth = 1;
            ctx.beginPath();
            ctx.moveTo(padLeft, y);
            ctx.lineTo(width - padRight, y);
            ctx.stroke();

            ctx.fillStyle = '#94a3b8';
            ctx.fillText(val + ' г', padLeft - 8, y);
        }

        currentChartPoints = rows.map((r, idx) => {
            const val = getValue(r);
            const xRatio = rows.length === 1 ? 0.5 : idx / (rows.length - 1);
            const yRatio = (val - minVal) / (maxVal - minVal || 1);
            const isHighlighted = filterKey === 'all' ? r.trained : r.groups.includes(filterKey);
            return {
                x: padLeft + xRatio * plotWidth,
                y: padTop + plotHeight - (yRatio * plotHeight),
                displayVal: val,
                isHighlighted: isHighlighted,
                data: r
            };
        });

        if (currentChartPoints.length > 1) {
            const grad = ctx.createLinearGradient(0, padTop, 0, padTop + plotHeight);
            grad.addColorStop(0, 'rgba(56, 189, 248, 0.30)');
            grad.addColorStop(1, 'rgba(56, 189, 248, 0.0)');

            ctx.beginPath();
            ctx.moveTo(currentChartPoints[0].x, padTop + plotHeight);
            currentChartPoints.forEach(p => ctx.lineTo(p.x, p.y));
            ctx.lineTo(currentChartPoints[currentChartPoints.length - 1].x, padTop + plotHeight);
            ctx.closePath();
            ctx.fillStyle = grad;
            ctx.fill();
        }

        ctx.beginPath();
        currentChartPoints.forEach((p, idx) => {
            if (idx === 0) ctx.moveTo(p.x, p.y);
            else ctx.lineTo(p.x, p.y);
        });
        ctx.strokeStyle = '#38bdf8';
        ctx.lineWidth = 2.5;
        ctx.stroke();

        ctx.fillStyle = '#94a3b8';
        ctx.textAlign = 'center';
        ctx.textBaseline = 'top';
        const labelStep = Math.max(1, Math.ceil(currentChartPoints.length / 6));

        currentChartPoints.forEach((p, idx) => {
            if (idx % labelStep === 0 || idx === currentChartPoints.length - 1) {
                const shortDate = p.data.date.slice(5);
                ctx.fillText(`Ден ${p.data.dayNumber} (${shortDate})`, p.x, padTop + plotHeight + 10);
            }
        });

        const pointRadius = currentChartPoints.length > 40 ? 3.5 : 5.5;
        currentChartPoints.forEach(p => {
            ctx.beginPath();
            ctx.arc(p.x, p.y, pointRadius, 0, Math.PI * 2);
            ctx.fillStyle = p.isHighlighted ? '#22c55e' : '#ef4444';
            ctx.fill();
            ctx.lineWidth = 1.5;
            ctx.strokeStyle = '#0f172a';
            ctx.stroke();
        });
    }

    function setupCanvasHover() {
        const canvas = document.getElementById('progressCanvas');
        const tooltip = document.getElementById('chartTooltip');

        canvas.addEventListener('mousemove', (e) => {
            if (currentChartPoints.length === 0) return;
            const rect = canvas.getBoundingClientRect();
            const mouseX = e.clientX - rect.left;
            const mouseY = e.clientY - rect.top;

            let closest = null;
            let minDist = Infinity;

            currentChartPoints.forEach(p => {
                const dist = Math.hypot(p.x - mouseX, p.y - mouseY);
                if (dist < minDist && dist < 35) {
                    minDist = dist;
                    closest = p;
                }
            });

            if (closest) {
                const d = closest.data;
                const boolColor = d.trained ? '#22c55e' : '#ef4444';
                const groupNames = d.groups.length > 0
                    ? d.groups.map(id => MUSCLE_GROUPS.find(g => g.id === id)?.name || id).join(', ')
                    : 'Общо възстановяване';

                tooltip.style.display = 'block';
                tooltip.innerHTML = `
                    <div style="font-weight:bold; margin-bottom:4px; color:#38bdf8;">
                        Ден #${d.dayNumber} (int) — ${d.date}
                    </div>
                    <div>Показана стойност (double): <strong>${closest.displayVal.toFixed(2)} г</strong></div>
                    <div>Прираст за деня: +${d.gainedGrams.toFixed(2)} г</div>
                    <div>Тренирал (bool): <span style="color:${boolColor}; font-weight:bold;">${d.trained}</span></div>
                    <div>Мускулни групи: <span style="color:#cbd5e1;">${groupNames}</span></div>
                `;

                let leftPos = closest.x + 12;
                if (leftPos + 230 > rect.width) {
                    leftPos = closest.x - 235;
                }
                let topPos = closest.y - 50;
                if (topPos < 5) topPos = 5;

                tooltip.style.left = leftPos + 'px';
                tooltip.style.top = topPos + 'px';
            } else {
                tooltip.style.display = 'none';
            }
        });

        canvas.addEventListener('mouseleave', () => {
            tooltip.style.display = 'none';
        });
    }

    function renderUI(db) {
        db.entries.sort((a, b) => a.date.localeCompare(b.date));

        const totalDaysInt = parseInt(db.entries.length, 10);

        // Инициализиране на сумите по мускулни групи (double за грамове, int за брой тренировки)
        const groupStats = {};
        MUSCLE_GROUPS.forEach(g => {
            groupStats[g.id] = {
                gramsDouble: 0.0,
                workoutsInt: 0
            };
        });

        let cumulativeDouble = 0.0;
        const rowsWithCumulative = db.entries.map((entry, index) => {
            const dayGrams = parseFloat(entry.gainedGrams) || 0.0;
            cumulativeDouble = parseFloat((cumulativeDouble + dayGrams).toFixed(2));
            const activeGroups = Array.isArray(entry.groups) ? entry.groups : [];

            if (activeGroups.length > 0) {
                // Разпределяме дневния грамаж между тренираните мускулни групи за деня
                const share = dayGrams / activeGroups.length;
                activeGroups.forEach(gId => {
                    if (groupStats[gId]) {
                        groupStats[gId].gramsDouble += share;
                        groupStats[gId].workoutsInt += 1;
                    }
                });
            } else {
                // Ако е почивен ден или няма избрана конкретна група, разпределяме базово между всички 6 групи
                const evenShare = dayGrams / MUSCLE_GROUPS.length;
                MUSCLE_GROUPS.forEach(g => {
                    groupStats[g.id].gramsDouble += evenShare;
                });
            }

            // Копие на текущите натрупани грамове по групи за графиката
            const snapshot = {};
            MUSCLE_GROUPS.forEach(g => {
                snapshot[g.id] = parseFloat(groupStats[g.id].gramsDouble.toFixed(2));
            });

            return {
                dayNumber: index + 1,
                date: entry.date,
                trained: Boolean(entry.trained),
                groups: activeGroups,
                gainedGrams: dayGrams,
                cumulativeGrams: cumulativeDouble,
                groupCumulative: snapshot
            };
        });

        const currentEntry = getCurrentEntry(db);
        const isTrainedTodayBool = currentEntry ? Boolean(currentEntry.trained) : false;
        const todayGroups = currentEntry && Array.isArray(currentEntry.groups) ? currentEntry.groups : [];

        // Основни 3 карти
        document.getElementById('valDouble').innerText = cumulativeDouble.toFixed(2) + ' г';
        const kg = (cumulativeDouble / 1000).toFixed(3);
        document.getElementById('valKg').innerText = `${kg} кг (темп: +${db.dailyGainRateGrams.toFixed(1)} г/ден)`;

        document.getElementById('valInt').innerText = totalDaysInt;
        document.getElementById('valDate').innerText = 'Последен запис: ' + (currentEntry ? currentEntry.date : '-');

        const boolEl = document.getElementById('valBool');
        const btnEl = document.getElementById('btnToggleBool');
        boolEl.innerText = isTrainedTodayBool ? 'true (Да)' : 'false (Не)';
        boolEl.style.color = isTrainedTodayBool ? '#22c55e' : '#ef4444';
        btnEl.innerText = isTrainedTodayBool ? 'Отмени за днес' : 'Отбележи тренировка';

        // Рендиране на 6-те мускулни групи
        const mgGrid = document.getElementById('muscleGroupsGrid');
        mgGrid.innerHTML = '';

        MUSCLE_GROUPS.forEach(g => {
            const stat = groupStats[g.id];
            const isTrainedToday = todayGroups.includes(g.id);
            const percent = cumulativeDouble > 0 ? Math.min(100, (stat.gramsDouble / cumulativeDouble) * 100) : 0;

            const card = document.createElement('div');
            card.className = 'muscle-card' + (isTrainedToday ? ' active-today' : '');
            card.innerHTML = `
                <div class="muscle-card-top">
                    <span class="muscle-name">${g.name}</span>
                    <span class="muscle-bool ${isTrainedToday ? 'status-true' : 'status-false'}">
                        bool: ${isTrainedToday}
                    </span>
                </div>
                <div class="muscle-stats">
                    <span class="muscle-grams" title="double">${stat.gramsDouble.toFixed(2)} г</span>
                    <span class="muscle-int" title="int">Тренировки (int): <strong>${stat.workoutsInt}</strong></span>
                </div>
                <div class="progress-bar-bg">
                    <div class="progress-bar-fill" style="width: ${percent.toFixed(1)}%;"></div>
                </div>
                <div style="display:flex; justify-content:space-between; align-items:center; margin-top:4px;">
                    <span class="subtext">${percent.toFixed(1)}% от общия мускул</span>
                    <button class="${isTrainedToday ? 'btn-outline' : ''} btn-sm" onclick="toggleTodayMuscleGroup('${g.id}')">
                        ${isTrainedToday ? 'Премахни днес' : '+ Тренирах днес'}
                    </button>
                </div>
            `;
            mgGrid.appendChild(card);
        });

        // Полета за настройки
        document.getElementById('inputDailyRate').value = db.dailyGainRateGrams;
        if (!document.getElementById('pastGramsInput').value) {
            document.getElementById('pastGramsInput').value = db.dailyGainRateGrams;
        }
        if (!document.getElementById('pastDateInput').value) {
            document.getElementById('pastDateInput').value = getTodayString();
        }
        renderPastGroupChips();

        // Рисуване на графиката спрямо избрания филтър
        const filterKey = document.getElementById('chartFilterSelect').value || 'all';
        drawChart(rowsWithCumulative, filterKey);

        // Обновяване на таблицата с история
        const tbody = document.getElementById('historyBody');
        tbody.innerHTML = '';

        rowsWithCumulative.slice().reverse().forEach(row => {
            const tr = document.createElement('tr');
            const groupBadges = row.groups.length > 0
                ? row.groups.map(id => `<span class="mini-tag">${MUSCLE_GROUPS.find(m => m.id === id)?.name || id}</span>`).join('')
                : `<span class="subtext">Базово (всички)</span>`;

            tr.innerHTML = `
                <td><strong>#${row.dayNumber}</strong></td>
                <td style="cursor:pointer; text-decoration:underline;" onclick="loadDateIntoEditor('${row.date}')" title="Кликни за редакция на този ден">${row.date}</td>
                <td>
                    <span class="status-badge ${row.trained ? 'status-true' : 'status-false'}" 
                          onclick="toggleWorkoutByDate('${row.date}')" 
                          title="Кликни за промяна на булевата стойност">
                        ${row.trained}
                    </span>
                </td>
                <td>${groupBadges}</td>
                <td>+${row.gainedGrams.toFixed(2)} г</td>
                <td><strong>${row.cumulativeGrams.toFixed(2)} г</strong></td>
                <td>
                    <button class="btn-outline btn-sm" onclick="loadDateIntoEditor('${row.date}')">Редактирай</button>
                    <button class="btn-danger btn-sm" onclick="deleteEntryByDate('${row.date}')">Изтрий</button>
                </td>
            `;
            tbody.appendChild(tr);
        });
    }

    window.addEventListener('resize', () => {
        const db = loadDB();
        renderUI(db);

    setupCanvasHover();
